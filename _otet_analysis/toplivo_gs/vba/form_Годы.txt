Private Sub OK_Click()
With Forms("Редактирование")!wform.Form
stcode = !numb1120
f = .filter
i1 = InStr(f, " and (year=")
If i1 = 0 Then GoTo Skip
i2 = InStr(i1, f, ")")
f = Left(f, i1 - 1) & Mid(f, i2 + 1)
Skip:
If Not годы.Selected(0) Then
fy = "("
For Each i In годы.ItemsSelected
If fy <> "(" Then fy = fy & " or "
fy = fy & "year=" & годы.Column(0, i)
Next i
fy = fy & ")"
'fy = fy & " or year=" & Form_Редактирование.byear & ")"
f = "(" & f & ") and " & fy
End If
.filter = f
.FilterOn = True
.Form.OrderBy = "numb1,year"
.OrderByOn = True
If InStr(f, "year") > 0 And InStr(f, Form_Редактирование.byear) = 0 Then
!name1.ColumnHidden = False
!Name.ColumnHidden = True
Else
!name1.ColumnHidden = True
!Name.ColumnHidden = False
End If
.RecordsetClone.FindFirst "numb1120=" & stcode
.Bookmark = .RecordsetClone.Bookmark
End With
End Sub
