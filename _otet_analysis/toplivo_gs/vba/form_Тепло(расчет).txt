Function ppf(y As Double)
Dim Q() As Double, R As Recordset
Set R = Me.RecordsetClone
ny = R.RecordCount
ReDim Q(ny - 1)
R.MoveFirst
If R!year = y Then
ppf = 0
GoTo endppf
End If
Do While (R!year <> y)
Qprev = R!Q
R.MoveNext
Loop
If Qprev <> 0 Then ppf = (R!Q - Qprev) / Qprev * 100 Else ppf = 0
endppf:
End Function
