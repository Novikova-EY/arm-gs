Private Sub Кнопка2_Click()
Dim years As DAO.Recordset
Stype = "АЭС,ГЭС,НВИЭ"
sf = "Eаэс,Eгэс,Eни"
atype = split(Stype, ",")
af = split(sf, ",")
sqly = "SELECT Дефициты.year FROM Дефициты GROUP BY Дефициты.year;"
Set years = CurrentDb.OpenRecordset(sqly)
years.MoveFirst
nxty:
y = years!year
For i = 0 To 2
sql1 = "UPDATE Дефициты INNER JOIN НеТЭС ON Дефициты.obl = НеТЭС.obl SET Дефициты.ffff = [НеТЭС]![2015]*1000"
sql1 = sql1 & " WHERE (((Дефициты.year)=2015) AND ((НеТЭС.type)=""tttt""));"
sql2 = Replace(sql1, "2015", Format(y, "####"))
sql2 = Replace(sql2, "tttt", atype(i))
sql2 = Replace(sql2, "ffff", af(i))
Debug.Print sql2
CurrentDb.Execute (sql2)
Next i
years.MoveNext
If Not (years.EOF) Then GoTo nxty
End Sub
