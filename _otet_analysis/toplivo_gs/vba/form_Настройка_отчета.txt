Private Sub Кнопка4_Click()
Dim db As Database, so As DAO.Recordset, work As QueryDef, sm As DAO.Recordset, sq As DAO.Recordset
Dim R As Report, swhere As String, join As String, w As QueryDef, D As QueryDef, i As Integer
Dim sqlwork As String, qtmp As QueryDef, selstan As QueryDef
Dim fr As QueryDef
Set db = Application.DBEngine.Workspaces(0).Databases(0)
Set sm = Список_макетов.Form.RecordsetClone
sm.Bookmark = Список_макетов.Form.Bookmark
l = 0
swhere = ""
If cond <> "" Then
swhere = " where " & cond
End If
If z(sm!U) = 1 Then GoTo skipquery
Set so = Список_объектов.Form.RecordsetClone
so.Bookmark = Список_объектов.Form.Bookmark
If z(sm!U) = 1.5 Then GoTo skipquery
Set sq = Список_запросов.Form.RecordsetClone
sq.Bookmark = Список_запросов.Form.Bookmark
If z(sm!U) = 2 Then GoTo skipquery
If so!vars <> "no" Then
If listvar <> "Все" Or listvar = "Все" And sq!sum = 1 Then
If swhere <> "" Then swhere = swhere & " and " Else swhere = "where "
If sq!sum <> 1 Then swhere = swhere & "(w.v=0 or w.v=" & listvar & ")" Else swhere = swhere & "(w.v=0 or w.v=1)"
End If
End If
Set work = db.QueryDefs(sq!Name)
Set fr = db.QueryDefs("для_отчета")
sqlwork = work.SQL
sqlwork = Replace(sqlwork, "Рабочий(станции)", so!Name)
sqlwork = Replace(sqlwork, "Рабочий(доп_угли)", so!opt)
If swhere <> "" Then sqlwork = Replace(sqlwork, "where", swhere & " and ")
If listvar = "Все" And sq!sum = 1 Then
vars = split(Replace(listvar.RowSource, ";Все", ""), ";")
sqlwork1 = sqlwork
sqlwork = Replace(sqlwork, "0 as v", "1 as v")
For i = 1 To UBound(vars)
sqlwork2 = Replace(sqlwork1, "0 as v", vars(i) & " as v")
sqlwork2 = Replace(sqlwork2, "v=1", "v=" & vars(i))
sqlwork = Replace(sqlwork, ";", " UNION " & sqlwork2)
Next i
End If
frold = fr.SQL
fr.SQL = sqlwork
DoCmd.OpenReport sm!Name, acViewPreview
skipquery:
If z(sm!U) = 2 Then
Set w = db.QueryDefs("Рабочий(станции)")
Set D = db.QueryDefs("Рабочий(доп_угли)")
sqlw1 = w.SQL
sqld1 = D.SQL
sqlw = "select w.* from ([#w] as w left join [#d] as d on (w.numb1120=d.numb1120) "
sqlw = sqlw & "and (w.year=d.year)) inner join имена_станций on w.numb1120=имена_станций.numb " & swhere & ";"
sqlw = Replace(sqlw, "#w", so!Name)
sqlw = Replace(sqlw, "#d", so!opt)
sqld = "select d.*,w.oes,w.obl from ([#w] as w inner join [#d] as d on (w.numb1120=d.numb1120) "
sqld = sqld & "and (w.year=d.year)) inner join имена_станций on w.numb1120=имена_станций.numb " & swhere & ";"
sqld = Replace(sqld, "#w", so!Name)
sqld = Replace(sqld, "#d", so!opt)
w.SQL = sqlw
D.SQL = sqld
End If
If z(sm!U) = 1.5 Then
Set work = db.QueryDefs(sm!Q)
sqlw = work.SQL
sqlwork = work.SQL
sqlwork = Replace(sqlwork, "Рабочий(станции)", so!Name)
sqlwork = Replace(sqlwork, "Рабочий(доп_угли)", so!opt)
If swhere <> "" Then sqlwork = Replace(sqlwork, "where", swhere & " and ")
Debug.Print sqlwork
work.SQL = sqlwork
End If
If z(sm!U) <> 1 Or (z(sm!U) = 1 And swhere = "") Then Application.DoCmd.OpenReport sm!Name, acViewPreview
If z(sm!U) = 1 And swhere <> "" Then
swhere = Replace(swhere, "w.", "")
Application.DoCmd.OpenReport sm!Name, acViewPreview, , Mid(swhere, 7)
'Reports(sm!name).filter = Mid(swhere, 7)
'Reports(sm!name).FilterOn = True
End If
'If l > 2048 Then db.QueryDefs.Delete ("tmp001")
If z(sm!U) = 2 Then
w.SQL = sqlw1
D.SQL = sqld1
End If
If z(sm!U) = 1.5 Then work.SQL = sqlw
If z(sm!U) <> 1 And z(sm!U) <> 1.5 And z(sm!U) <> 2 Then fr.SQL = frold
End Sub
