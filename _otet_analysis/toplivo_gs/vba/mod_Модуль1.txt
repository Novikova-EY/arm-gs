Public Function repbetw(s As Variant, s1 As String, s2 As String, s3 As Variant) As Variant
i2 = InStr(1, s, s2, vbTextCompare)
i1 = InStrRev(Mid(s, 1, i2), s1, , vbTextCompare)
repbetw = Mid(s, 1, i1 - 1 + Len(s1)) & s3 & Mid(s, i2)
End Function
Public Function repbetw1(s As Variant, s1 As String, s2 As String, s3 As Variant) As Variant
i1 = InStr(1, s, s1, vbTextCompare)
i2 = InStr(1, Mid(s, i1 + Len(s1)), s2, vbTextCompare)
repbetw1 = Left(s, i1 + Len(s1) - 1) & s3 & Mid(s, i1 + Len(s1) + i2 - 1)
End Function

If (Bk = bkcp) Then Interval = 3
