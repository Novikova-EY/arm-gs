Private Sub Form_AfterInsert()
Dim trs As DAO.Recordset, otetopes As DAO.Recordset
Set trs = CurrentDb.OpenRecordset("Списки_агрегатов")
Set otetopes = CurrentDb.OpenRecordset("Имена_станций")
otetopes.FindFirst "NUMB=" & Me!stcode
msg = InputBox("Введите установленную мощность")
NUST = msg
For i = 0 To Me.Controls.Count - 1
If Me.Controls(i).Tag = "y" Then
If CDbl(Me.Controls(i).Name) >= CDbl(Me!year) Then Me.Controls(i) = NUST
'Me!stcodeopes = otetopes("ОПЭС")
End If
Next i
If ur = 1 Then
trs.AddNew
For i = 0 To Me.Controls.Count - 1
trs(Controls(i).Name) = Controls(i)
trs("ur") = 2
Next i
trs.Update
End If
'DoCmd.RunSQL (CurrentDb.QueryDefs("код_ОПЭС_в_списки").SQL)
End Sub

ur = 1
End With
End Sub

Dim tform As Form
'Set tform = Me.Parent

Next i
