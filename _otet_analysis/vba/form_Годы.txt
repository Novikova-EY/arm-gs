Private Sub OK_Click()
With Forms("Редактирование_отчетных_данных")!stan.Form
f = .Filter
i1 = InStr(f, " and (year=")
If i1 = 0 Then GoTo skip
i2 = InStr(i1, f, ")")
f = left(f, i1 - 1) & Mid(f, i2 + 1)
skip:
If Not годы.Selected(0) Then
fy = "("
For Each i In годы.ItemsSelected
If fy <> "(" Then fy = fy & " or "
fy = fy & "year=" & годы.Column(0, i)
Next i
fy = fy & ")"
f = f & " and " & fy
End If
.Filter = f
.FilterOn = True
.Form.OrderBy = "numb1,year"
.OrderByOn = True
End With
End Sub
