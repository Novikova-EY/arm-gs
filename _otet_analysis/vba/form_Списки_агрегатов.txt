Sub stnumbnotfound(n As Double)
sendmsg ("агрегат не найден")
End Sub
Function matchn(gnt As Variant, ant As Variant) As Boolean
matchn = False
If IsNull(gnt) Then
matchn = True
Else
it = InStr(gnt, "-")
If it = 0 Then
nt1 = CDbl(gnt)
nt2 = CDbl(gnt)
Else
nt1 = CDbl(left(gnt, it - 1))
nt2 = CDbl(Mid(gnt, it + 1))
End If
If ant = "..." Then matchn = True Else If CDbl(ant) >= nt1 And CDbl(ant) <= nt2 Then matchn = True
End If
End Function
Function matchp(gpt As Variant, apt As Variant) As Boolean
matchp = False
If IsNull(gpt) Then
matchp = True
Else
If left(gpt, 2) = "<=" Then
pt1 = 0
pt2 = CDbl(Mid(gpt, 3))
Else
pt1 = CDbl(gpt)
pt2 = CDbl(gpt)
End If
If CDbl(apt) >= pt1 And CDbl(apt) <= pt2 Then matchp = True
End If
End Function
Sub groupfound(stcode As Double) 'Группа найдена
gfound = True
years = Array("2000", "2001", "2002", "2003", "2004", "2005", "2010", "2015", "2020")
For i = 0 To UBound(years)
out.Seek "=", stcode, years(i)
If out.NoMatch Then
out.AddNew
out!NUMB1120 = stcode
out!YEAR = years(i)
Else: out.Edit
End If
out!ne = z(out!ne) + z(lista(years(i)))
If z(lista(years(i))) > 0 Then out!NT = z(out!NT) + antep
If lista!stcode = 27 Then Debug.Print antep; out!NT
out!p = 1
out.Update
Next i
End Sub

Function matcht(gstt As Variant, astt As Variant) As Boolean
matcht = False
If IsNull(gstt) Then matcht = True
If gstt = "КЭС" And astt = "К" Then matcht = True
If gstt = "ТЭЦ" And InStr("Т;ПТ;П;Р;ПР;ТЭЦ", astt) > 0 Then matcht = True
If gstt = "ПСУ" And astt <> "ПГУ" And astt <> "ГТ" Then matcht = True
If gstt = "ПГУ" And InStr(astt, "ПГУ") > 0 Then matcht = True
If gstt = "ГТ" And InStr(astt, "ГТ") > 0 Then matcht = True
End Function
Sub sendmsg(txt As String)

msgf!NUMB1120 = lista!stcode

End Sub
